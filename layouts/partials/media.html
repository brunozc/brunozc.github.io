A missing file renders a placeholder box during `hugo server` and nothing in the production build, so figures can be dropped into static/ later without touching the templates. */ -}} {{- $src := strings.TrimPrefix "/" .src }} {{- $file := path.Join "static" $src }} {{- $ext := lower (path.Ext $src) }} {{- $alt := .alt | default .caption | default "" | plainify }} {{- if fileExists $file }} {{- else if not hugo.IsProduction }} {{- end }}
